1、Internet Explorer ​

  • 脚本如下
shell
Rem 这是一个vbs脚本

Const URL_TO_OPEN = "https://www.baidu.com"
Dim objIE
Set objIE = CreateObject("InternetExplorer.Application")
objIE.Visible = True
objIE.Navigate URL_TO_OPEN

Do While objIE.Busy Or objIE.ReadyState <> 4
    WScript.Sleep 100
Loop

Set objIE = Nothing

2、删除RDP记录 ​

  • 脚本如下
shell
@echo off
chcp 65001 > nul
setlocal enabledelayedexpansion
title 删除RDP记录

:: 检查是否以管理员身份运行
openfiles >nul 2>&1
if %errorlevel% neq 0 (
    echo 正在以管理员权限运行...
    powershell -Command "Start-Process '%~f0' -Verb runAs"
    exit /b
)
  
reg delete “HKEY_CURRENT_USER\Software\Microsoft\Terminal Server Client\Default” /va /f
reg delete “HKEY_CURRENT_USER\Software\Microsoft\Terminal Server Client\Servers” /f
reg add “HKEY_CURRENT_USER\Software\Microsoft\Terminal Server Client\Servers”
del /ah %homepath%\documents\default.rdp

timeout /t 3

3、关闭推荐的项目 ​

  • 脚本如下
shell
@echo off
chcp 65001 > nul
setlocal enabledelayedexpansion
title Win11推荐项目设置器 v1.0

:: 检查是否以管理员身份运行
openfiles >nul 2>&1
if %errorlevel% neq 0 (
    echo 正在以管理员权限运行...
    powershell -Command "Start-Process '%~f0' -Verb runAs"
    exit /b
)

:menu
cls
echo.
echo ============================================
echo     Win11推荐项目设置器 v1.0
echo ============================================
echo.
echo  1. 关闭“推荐的项目”（隐藏“推荐”区域）
echo.
echo  2. 启用“推荐的项目”（恢复默认显示）
echo.
echo  3. 重启资源管理器（快速应用更改）
echo.
echo  0. 退出
echo.
set /p choice= 请输入选项对应的数字 [0-3]，按回车键确认：

if "%choice%"=="1" goto disable
if "%choice%"=="2" goto enable
if "%choice%"=="3" goto restart
if "%choice%"=="0" exit
goto menu

:disable
echo 正在关闭推荐的项目...
reg add "HKLM\SOFTWARE\Policies\Microsoft\Windows\Explorer" /v HideRecommendedSection /t REG_DWORD /d 1 /f
reg add "HKLM\SOFTWARE\Microsoft\PolicyManager\current\device\Start" /v HideRecommendedSection /t REG_DWORD /d 1 /f
reg add "HKLM\SOFTWARE\Microsoft\PolicyManager\current\device\Education" /v IsEducationEnvironment /t REG_DWORD /d 1 /f
echo 操作完成。
pause
goto menu

:enable
echo 正在启用推荐的项目...
reg delete "HKLM\SOFTWARE\Policies\Microsoft\Windows\Explorer" /v HideRecommendedSection /f
reg delete "HKLM\SOFTWARE\Microsoft\PolicyManager\current\device\Start" /v HideRecommendedSection /f
reg delete "HKLM\SOFTWARE\Microsoft\PolicyManager\current\device\Education" /v IsEducationEnvironment /f
echo 操作完成。
pause
goto menu

:restart
echo 正在重启资源管理器...
taskkill /f /im explorer.exe >nul 2>&1
timeout /t 1 >nul
start explorer.exe
echo 资源管理器已重启，可重新打开开始菜单，查看设置是否生效。
pause
goto menu

5、内核隔离管理工具 ​

  • 脚本如下
shell
@echo off
chcp 65001 > nul
setlocal enabledelayedexpansion
title 内核隔离管理工具

:: 检查是否以管理员身份运行
openfiles >nul 2>&1
if %errorlevel% neq 0 (
    echo 正在以管理员权限运行...
    powershell -Command "Start-Process '%~f0' -Verb runAs"
    exit /b
)

:menu
cls
echo ========================================
echo           内核隔离管理工具
echo ========================================
echo.
echo   基于您之前提供的注册表命令：
echo.
echo   1. 关闭内核隔离
echo   2. 启用内核隔离
echo   3. 退出脚本
echo.
echo ========================================

set /p choice=请选择操作 (1/2/3): 

if "%choice%"=="1" goto disable_vbs
if "%choice%"=="2" goto enable_vbs  
if "%choice%"=="3" goto exit
echo 无效选择，请按任意键重新选择...
pause > nul
goto menu

:disable_vbs
echo.
echo 正在关闭内核隔离...
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard" /v EnableVirtualizationBasedSecurity /t REG_DWORD /d 0 /f
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" /v Enabled /t REG_DWORD /d 0 /f
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" /v EnabledBootId /t REG_DWORD /d 0 /f
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" /v WasEnabledBy /t REG_DWORD /d 1 /f
echo 内核隔离已关闭！
goto restart

:enable_vbs
echo.
echo 正在启用内核隔离...
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard" /v EnableVirtualizationBasedSecurity /t REG_DWORD /d 1 /f
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" /v Enabled /t REG_DWORD /d 1 /f
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" /v EnabledBootId /t REG_DWORD /d 0 /f
reg add "HKLM\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" /v WasEnabledBy /t REG_DWORD /d 1 /f
echo 内核隔离已启用！
goto restart

:restart
echo.
echo 操作已完成！计算机将在60秒后重启...
echo 如需取消重启，请按 Ctrl+C 并输入 shutdown /a
echo.
shutdown /r /t 60 /c "内核隔离设置已完成，系统将重启以生效"
pause
goto exit

:exit
echo 脚本执行完毕！
pause

11、Nginx操作脚本 ​

  • 脚本如下
shell
@echo off
setlocal enabledelayedexpansion
chcp 65001 >nul
title Nginx 管理工具

:: ================= 配置信息 =================
:: 如果脚本放在 Nginx 根目录下，保持 %~dp0 即可
set "NGINX_PATH=%~dp0"
set "NGINX_EXE=nginx.exe"
set "URL=http://127.0.0.1"
:: ============================================

:MENU
cls
echo ======================================================
echo       Nginx 增强管理器 (当前目录: %NGINX_PATH%)
echo ======================================================
echo  [1] 启动 Nginx 并访问网页 (含冲突检测)
echo  [2] 平滑停止 Nginx
echo  [3] 检查并重载配置
echo  [4] 强制杀掉所有 Nginx 进程
echo  [5] 查看运行状态与端口占用
echo  [6] 打开错误日志
echo  [7] 清理并退出
echo ======================================================
set /p choice=请输入编号并回车: 

if "%choice%"=="1" goto START
if "%choice%"=="2" goto STOP
if "%choice%"=="3" goto RELOAD
if "%choice%"=="4" goto KILL
if "%choice%"=="5" goto STATUS
if "%choice%"=="6" goto LOG
if "%choice%"=="7" goto KILL_AND_EXIT
goto MENU

:START
echo.
echo [检查] 正在验证环境...
if not exist "%NGINX_PATH%\%NGINX_EXE%" (
    echo [错误] 找不到 %NGINX_EXE%，请确认脚本放置在 Nginx 根目录。
    goto KILL_AND_EXIT
)

tasklist | findstr /I "%NGINX_EXE%" >nul
if %ERRORLEVEL% equ 0 (
    echo [提示] Nginx 已经在运行中，正在为您打开浏览器...
    start %URL%
    timeout /t 2 >nul
    goto MENU
)

cd /d "%NGINX_PATH%"
start "" "%NGINX_EXE%"
timeout /t 2 /nobreak >nul

tasklist | findstr /I "%NGINX_EXE%" >nul
if %ERRORLEVEL% equ 0 (
    echo [成功] Nginx 已启动。
    start %URL%
) else (
    echo [失败] Nginx 未能启动！
    echo [分析] 可能是 80 端口被占用，正在检测端口状态:
    netstat -ano | findstr ":80"
    echo [建议] 请检查上方显示的 PID 进程，或尝试使用 [4] 强行清理。
    timeout /t 5 >nul
)
goto MENU

:STOP
echo.
echo [操作] 正在发送平滑停止信号...
cd /d "%NGINX_PATH%"
"%NGINX_EXE%" -s quit
if %ERRORLEVEL% equ 0 (
    echo [完成] Nginx 已安全退出。
) else (
    echo [提示] 停止命令执行异常，可能进程本就未启动。
)
timeout /t 2 >nul
goto MENU

:RELOAD
echo.
echo [检查] 正在校验配置文件语法...
cd /d "%NGINX_PATH%"
"%NGINX_EXE%" -t
if %ERRORLEVEL% equ 0 (
    echo [成功] 语法正确，正在重载配置...
    "%NGINX_EXE%" -s reload
    start %URL%
) else (
    echo [严重] 语法检测失败！请根据上方提示修改配置后再试。
    pause
)
goto MENU

:KILL
echo.
echo [操作] 强制清理所有 Nginx 进程...
taskkill /F /IM %NGINX_EXE% /T >nul 2>&1
echo [完成] 所有相关进程已强制结束。
timeout /t 2 >nul
goto MENU

:STATUS
echo.
echo [状态] 当前 Nginx 进程列表:
tasklist | findstr /I "%NGINX_EXE%" || echo 未发现运行中的进程。
echo.
echo [端口] 本地 80/443 端口监听情况:
netstat -ano | findstr /R ":80 :443"
pause
goto MENU

:LOG
echo.
echo [操作] 正在定位日志资源...
cd /d "%NGINX_PATH%"

if exist "logs\error.log" (
    echo [成功] 正在使用记事本打开 error.log...
    start notepad "logs\error.log"
    timeout /t 2 >nul
) else (
    echo [提示] 未找到具体的 error.log 文件。
    if exist "logs\" (
        echo [动作] 正在为您打开 logs 文件夹进行检查...
        start explorer "logs\"
    ) else (
        echo [错误] 连 logs 文件夹也不存在，请确认 Nginx 是否运行过。
        pause
    )
)
goto MENU

:KILL_AND_EXIT
echo.
echo [操作] 正在强制清理 Nginx 进程并退出...
taskkill /F /IM %NGINX_EXE% /T >nul 2>&1
echo [完成] 进程已清理，回车关闭窗口。
timeout /t 2 >nul
exit
